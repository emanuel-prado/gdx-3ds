# Standalone configure: no game, Fast3D, archives, bridge, or simulation objects.
cmake_minimum_required(VERSION 3.20)
project(gdx_native_m1a C)
if(NOT NINTENDO_3DS)
    message(FATAL_ERROR "Use the devkitPro 3DS toolchain")
endif()
ctr_add_shader_library(native_shader OUTPUT native_shader.shbin "${CMAKE_CURRENT_SOURCE_DIR}/shader.v.pica")
dkp_add_embedded_binary_library(native_shbin native_shader)
add_executable(gdx-native-m1a main.c pack.c)
execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.c" MAIN_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/pack.c" PACK_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/shader.v.pica" SHADER_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/pack.h" HEADER_SHA)
string(SHA256 SOURCE_ID "${MAIN_SHA}${PACK_SHA}${SHADER_SHA}${HEADER_SHA}")
execute_process(COMMAND dkp-pacman -Q devkitARM libctru citro3d 3dstools OUTPUT_VARIABLE TOOLCHAIN_ID OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\n" " / " TOOLCHAIN_ID "${TOOLCHAIN_ID}")
target_compile_definitions(gdx-native-m1a PRIVATE NATIVE_BUILD_ID="${BUILD_ID}" NATIVE_SOURCE_ID="${SOURCE_ID}" NATIVE_TOOLCHAIN_ID="${TOOLCHAIN_ID}")
target_compile_options(gdx-native-m1a PRIVATE -Wall -Wextra -Werror)
target_link_libraries(gdx-native-m1a PRIVATE native_shbin citro3d ctru m)
ctr_create_3dsx(gdx-native-m1a)
