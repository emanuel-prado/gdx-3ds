# Standalone configure: no game, Fast3D, archives, bridge, or simulation objects.
cmake_minimum_required(VERSION 3.20)
project(gdx_native_m1a C)
if(NOT NINTENDO_3DS)
    message(FATAL_ERROR "Use the devkitPro 3DS toolchain")
endif()
ctr_add_shader_library(native_shader OUTPUT native_shader.shbin "${CMAKE_CURRENT_SOURCE_DIR}/shader.v.pica")
dkp_add_embedded_binary_library(native_shbin native_shader)
add_executable(gdx-native-m1a main.c pack.c)
set(NATIVE_BUILD_ID_OVERRIDE "" CACHE STRING "Explicit source revision when linked-worktree git metadata is outside container")
execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NATIVE_BUILD_ID_OVERRIDE)
    set(BUILD_ID "${NATIVE_BUILD_ID_OVERRIDE}")
elseif(NOT BUILD_ID)
    set(BUILD_ID "unavailable-isolated-worktree")
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/main.c" MAIN_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/pack.c" PACK_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/shader.v.pica" SHADER_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/pack.h" HEADER_SHA)
string(SHA256 SOURCE_ID "${MAIN_SHA}${PACK_SHA}${SHADER_SHA}${HEADER_SHA}")
execute_process(COMMAND dkp-pacman -Q devkitARM libctru citro3d 3dstools OUTPUT_VARIABLE TOOLCHAIN_ID OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\n" " / " TOOLCHAIN_ID "${TOOLCHAIN_ID}")
target_compile_definitions(gdx-native-m1a PRIVATE NATIVE_BUILD_ID="${BUILD_ID}" NATIVE_SOURCE_ID="${SOURCE_ID}" NATIVE_TOOLCHAIN_ID="${TOOLCHAIN_ID}")
target_compile_options(gdx-native-m1a PRIVATE -Wall -Wextra -Werror)
target_link_libraries(gdx-native-m1a PRIVATE native_shbin citro3d ctru m)
ctr_create_3dsx(gdx-native-m1a)

ctr_add_shader_library(swatch_shader OUTPUT swatch_shader.shbin "${CMAKE_CURRENT_SOURCE_DIR}/swatch.v.pica")
dkp_add_embedded_binary_library(swatch_shbin swatch_shader)
add_executable(gdx-native-swatch swatch.c texture_pack.c)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/swatch.c" SWATCH_MAIN_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/texture_pack.c" SWATCH_PACK_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/texture_pack.h" SWATCH_HEADER_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/swatch.v.pica" SWATCH_SHADER_SHA)
string(SHA256 SWATCH_SOURCE_ID "${SWATCH_MAIN_SHA}${SWATCH_PACK_SHA}${SWATCH_HEADER_SHA}${SWATCH_SHADER_SHA}")
target_compile_definitions(gdx-native-swatch PRIVATE NATIVE_BUILD_ID="${BUILD_ID}" NATIVE_SOURCE_ID="${SWATCH_SOURCE_ID}" NATIVE_TOOLCHAIN_ID="${TOOLCHAIN_ID}")
target_compile_options(gdx-native-swatch PRIVATE -Wall -Wextra -Werror)
target_link_libraries(gdx-native-swatch PRIVATE swatch_shbin citro3d ctru m)
ctr_create_3dsx(gdx-native-swatch)

ctr_add_shader_library(material_shader OUTPUT material_shader.shbin "${CMAKE_CURRENT_SOURCE_DIR}/material.v.pica")
dkp_add_embedded_binary_library(material_shbin material_shader)
add_executable(gdx-native-materials material.c material_scene.c pack.c texture_pack.c)
set(MATERIAL_SOURCE_PARTS "")
foreach(MATERIAL_FILE material.c material_scene.c material_scene.h material.v.pica pack.c pack.h texture_pack.c texture_pack.h)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${MATERIAL_FILE}" MATERIAL_FILE_SHA)
    string(APPEND MATERIAL_SOURCE_PARTS "${MATERIAL_FILE_SHA}")
endforeach()
string(SHA256 MATERIAL_SOURCE_ID "${MATERIAL_SOURCE_PARTS}")
target_compile_definitions(gdx-native-materials PRIVATE NATIVE_BUILD_ID="${BUILD_ID}" NATIVE_SOURCE_ID="${MATERIAL_SOURCE_ID}" NATIVE_TOOLCHAIN_ID="${TOOLCHAIN_ID}")
target_compile_options(gdx-native-materials PRIVATE -Wall -Wextra -Werror)
target_link_libraries(gdx-native-materials PRIVATE material_shbin citro3d ctru m)
ctr_create_3dsx(gdx-native-materials)
